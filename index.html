// ========================================== // 1. TẠO MENU TÙY CHỈNH TRÊN GOOGLE SHEETS // ========================================== function onOpen() { var ui = SpreadsheetApp.getUi(); ui.createMenu('🛠️ HỆ THỐNG QUẢN LÝ QR') .addItem('🚀 Khởi tạo tự động 2 Sheet (Bandemo & BanFull)', 'setupSheetsAndHeaders') .addSeparator() .addItem('📥 Tạo File QR cho dòng đang chọn', 'generateQRFile') .addToUi(); } // ========================================== // 2. TỰ ĐỘNG TẠO SHEET VÀ TIÊU ĐỀ CỘT // ========================================== function setupSheetsAndHeaders() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheetConfigs = [ { name: "Bandemo", title: "QUẢN LÝ BẢN DEMO 30 PHÚT" }, { name: "BanFull", title: "QUẢN LÝ BẢN FULL CHÍNH THỨC" } ]; var headers = ["Key", "Tên người dùng / Đơn vị", "Ngày bắt đầu", "Thời hạn (Tháng)", "Ngày kết thúc", "Trạng thái", "DeviceID"]; sheetConfigs.forEach(function(config) { var sheet = ss.getSheetByName(config.name); if (!sheet) { sheet = ss.insertSheet(config.name); } // Kiểm tra xem sheet đã có tiêu đề chưa, nếu chưa thì điền vào dòng 1 và 2 var firstCell = sheet.getRange(1, 1).getValue(); if (!firstCell) { sheet.getRange(1, 1).setValue(config.title); sheet.getRange(1, 1).setFontWeight("bold").setFontSize(14); // Ghi tiêu đề cột ở dòng 2 sheet.getRange(2, 1, 1, headers.length).setValues([headers]); sheet.getRange(2, 1, 1, headers.length).setFontWeight("bold").setBackground("#334155").setFontColor("#ffffff"); // Định dạng độ rộng cột cho đẹp sheet.setColumnWidth(1, 180); // Key sheet.setColumnWidth(2, 220); // Tên sheet.setColumnWidth(3, 130); // Ngày bắt đầu sheet.setColumnWidth(4, 120); // Thời hạn sheet.setColumnWidth(5, 130); // Ngày kết thúc sheet.setColumnWidth(6, 100); // Trạng thái sheet.setColumnWidth(7, 200); // DeviceID } }); SpreadsheetApp.getUi().alert("✅ Đã khởi tạo và cấu hình thành công 2 sheet Bandemo và BanFull!"); } // ========================================== // 3. TẠO FILE QR CHO DÒNG ĐANG CHỌN (GỬI LINK QUÉT KÈM TYPE) // ========================================== function generateQRFile() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var sheetName = sheet.getName(); if (sheetName !== "Bandemo" && sheetName !== "BanFull") { SpreadsheetApp.getUi().alert("⚠️ Vui lòng click chọn dòng trong sheet Bandemo hoặc BanFull!"); return; } var row = sheet.getActiveCell().getRow(); if (row < 3) { SpreadsheetApp.getUi().alert("⚠️ Vui lòng chọn một dòng chứa mã Key (từ dòng 3 trở đi, sau hàng tiêu đề)!"); return; } var key = String(sheet.getRange(row, 1).getValue()).trim(); // Cột A: Key var tenNguoiDung = String(sheet.getRange(row, 2).getValue()).trim(); // Cột B: TenNguoiDung if (!key) { SpreadsheetApp.getUi().alert("⚠️ Dòng này chưa có mã Key!"); return; } var ui = SpreadsheetApp.getUi(); ui.alert("⏳ Đang tạo file Google Docs chứa mã QR..."); try { // THAY ĐƯỜNG DẪN WEB APP URL CỦA THẦY VÀO ĐÂY (Đuôi /exec) var WEB_APP_URL = "YOUR_WEB_APP_URL_O_DAY"; var licenseType = (sheetName === "Bandemo") ? "demo" : "full"; var targetUrl = WEB_APP_URL + "?key=" + key + "&type=" + licenseType; var qrImageUrl = "https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=" + encodeURIComponent(targetUrl); var docName = "QR_" + licenseType.toUpperCase() + "_" + key; var doc = DocumentApp.create(docName); var body = doc.getBody(); body.appendHeading1("MÃ QR KÍCH HOẠT DAO ĐỘNG KÝ (" + licenseType.toUpperCase() + ")"); body.appendParagraph("Người dùng / Đơn vị: " + tenNguoiDung); body.appendParagraph("Phân loại: " + (licenseType === "demo" ? "Dùng thử 30 Phút" : "Bản Full (Chính thức)")); body.appendParagraph("Mã Key bảo mật: " + key); body.appendParagraph("--------------------------------------------------"); var imageBlob = UrlFetchApp.fetch(qrImageUrl).getBlob(); var img = body.appendImage(imageBlob); img.setWidth(300); img.setHeight(300); body.appendParagraph("--------------------------------------------------"); body.appendParagraph("📌 Hướng dẫn: Quét mã QR này bằng điện thoại để kích hoạt. Hệ thống sẽ tự động khóa cứng vào thiết bị đầu tiên quét."); doc.saveAndClose(); ui.alert("✅ Tạo file QR thành công!\n\nTên file: " + docName + "\nĐã lưu trong Google Drive của bạn."); } catch (err) { ui.alert("❌ Lỗi: " + err.toString()); } } // ========================================== // 4. XỬ LÝ XÁC THỰC TỪ ĐIỆN THOẠI VÀ GHI NHẬN ĐÚNG 2 SHEET // ========================================== function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var action = data.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "verifyKey") { var userKey = String(data.key).trim(); var clientDeviceId = String(data.deviceId).trim(); var requestedType = String(data.type || "demo").trim().toLowerCase(); var targetSheetName = (requestedType === "full") ? "BanFull" : "Bandemo"; var sheet = ss.getSheetByName(targetSheetName); if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Không tìm thấy hệ thống sheet tương ứng!" })).setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getDataRange().getValues(); var isGranted = false; var message = "Mã Key không hợp lệ trong hệ thống " + targetSheetName + "!"; var now = new Date(); // Bắt đầu duyệt từ dòng 3 (bỏ qua tiêu đề ở dòng 1 và 2) for (var i = 2; i < rows.length; i++) { var dbKey = String(rows[i][0]).trim(); // Cột A: Key var startDateVal = rows[i][2]; // Cột C: NgayBatDau var monthsToAdd = parseInt(rows[i][3]) || 0; // Cột D: ThoiHanThang var endDateVal = rows[i][4]; // Cột E: NgayKetThuc var dbCheckbox = rows[i][5]; // Cột F: TrangThai (Checkbox) if (dbKey === userKey) { var startDate = new Date(startDateVal || now); var expireDate; if (endDateVal && endDateVal !== "" && endDateVal !== "Vĩnh viễn") { expireDate = new Date(endDateVal); } else if (monthsToAdd > 0) { expireDate = new Date(startDate); expireDate.setMonth(expireDate.getMonth() + monthsToAdd); } else { expireDate = new Date(startDate); if (requestedType === "demo") { expireDate.setMinutes(expireDate.getMinutes() + 30); } else { expireDate.setFullYear(expireDate.getFullYear() + 100); } } if (now > expireDate) { var rowNum = i + 1; sheet.getRange(rowNum, 6).setValue(false); sheet.getRange(rowNum, 5).setBackground("#f87171").setFontColor("#ffffff"); message = "Mã Key đã hết hạn sử dụng!"; break; } if (dbCheckbox !== true && dbCheckbox !== "TRUE") { message = "Mã Key này đang ở trạng thái BỊ KHÓA!"; break; } // Khóa cứng 1 Key cho 1 máy dựa trên Cột G (Cột số 7) var deviceIdCol = 7; var dbDeviceId = String(rows[i][6] || "").trim(); if (dbDeviceId === "") { sheet.getRange(i + 1, deviceIdCol).setValue(clientDeviceId); isGranted = true; message = "Xác thực và khóa cứng thiết bị thành công (" + targetSheetName + ")!"; } else if (dbDeviceId === clientDeviceId) { isGranted = true; message = "Xác thực thiết bị hợp lệ!"; } else { message = "⛔ LỖI: Mã Key này đã được kích hoạt trên một thiết bị khác!"; } break; } } return ContentService.createTextOutput(JSON.stringify({ "status": isGranted ? "success" : "error", "message": message, "type": requestedType })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
